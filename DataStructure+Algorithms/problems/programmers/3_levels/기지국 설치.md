### 경계 조건 
질문: "예외나 특수한 상황은 어디에 존재하는가?"

    1. 첫 번째 기지국 전의 공간(1번 ~ 첫 기지국 시작점), 
    2. 마지막 기지국 후의 공간(마지막 기지국 끝점 ~ N)
    
    일반적인 기지국 사이 공간과 동일한 로직으로 처리되는지 검증필요.
    
### 최적화 알고리즘 결정

질문: "그리디(Greedy), DP, 이진 탐색 중 왜 하필 이 방식인가?"

앞에서부터 차례대로 필요한 만큼만 채워나가는 방식(Greedy)이 항상 최적해를 보장



# 코드

    class Solution {
        public int solution(int n, int[] stations, int w) {
            int answer = 0;

            // 추상화 - 기지국 1개가 커버할 수 있는 전체 커버리지 단위
            int coverage = 2 * w + 1;

            //  경계 조건 , 추상화-  아직 전파가 도달하지 않은 구간의 시작점 (1번 아파트부터 시작)
            int position = 1;

            //  제약 조건 파악- N을 순회하지 않고, stations 순회하여 O(M)으로 해결
            for (int station : stations) {
                // 현재 기지국이 커버하는 전파의 좌측 끝점
                int startCovered = station - w;

                // 만약 현재 커버리지 시작점보다 position이 앞에 있다면, 그 사이에 '빈 구간'이 존재함
                if (position < startCovered) {
                    int gap = startCovered - position; // 빈 구간의 길이

                    //  최적화 알고리즘 - Greedy 빈 구간에 필요한 최소 기지국 개수 계산 및 가산
                    answer += gap / coverage;
                    if (gap % coverage != 0) {
                        answer++; // 나누어 떨어지지 않으면 1개 더 필요
                    }
                }

                //  경계 조건 - 다음 빈 구간의 시작점 업데이트 (현재 기지국이 커버하는 우측 끝점 + 1)
                position = station + w + 1;
            }

            //경계 조건 - 마지막 기지국 설치 이후~전체 아파트 끝(N)까지 남은 빈 구간 처리
            if (position <= n) {
                int gap = n - position + 1;

                answer += gap / coverage;
                if (gap % coverage != 0) {
                    answer++;
                }
            }

            return answer;
        }
    }

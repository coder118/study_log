# 문제 정의

### 제약 분석

- 입력 문자열 p의 길이 $N$은 2 이상 1,000 이하의 짝수
- O(N^2) 내외의 알고리즘으로도 충분히 통과

### 패턴 단순 추상화

1. 균형잡힌 문자열 분리 (u, v 분리)

2. 올바른 괄호 문자열 검증

3. 재귀적 변환 알고리즘

### 경계 조건

- 빈 문자열("") 입력 - 입력받은 문자열이 빈 문자열일 경우, 빈 문자열을 그대로 반환하고 재귀를 종료

- v가 빈 문자열인 경우:

u가 전체 문자열과 같아서 v가 ""가 되는 경우도, 지문의 재귀 로직(1단계 반환)에 의해 자연스럽게 처리

- u의 첫 번째와 마지막 문자를 제거한 뒤 Remaining string이 빈 문자열인 경우:

u의 길이가 2(예: ")(")인 경우, 양 끝 문자를 제거하면 빈 문자열이 된다.

- 괄호 방향 뒤집기 logic:

문자열 전체를 거꾸로 뒤집는 것이 아니라, '('는 ')'로, ')'는 '('로 괄호의 모양을 뒤집는 것을 생각


### 최적의 알고리즘

제시된 명확한 변환 규칙에 따라 문자열을 구성해나가는 "구현/시뮬레이션" 문제, 재귀 방식


# 코드


    class Solution {
        public String solution(String p) {
            // 1. 입력이 빈 문자열인 경우, 빈 문자열을 반환합니다.
            if (p.isEmpty()) {
                return "";
            }
            
            // 2. 문자열 w(p)를 두 "균형잡힌 괄호 문자열" u, v로 분리합니다.
            int splitIndex = getSplitIndex(p);
            String u = p.substring(0, splitIndex);
            String v = p.substring(splitIndex);
            
            // 3. 문자열 u가 "올바른 괄호 문자열" 이라면
            if (isCorrect(u)) {
                // 3-1.v에 대해 1단계부터 다시 수행한 결과를 u에 이어 붙인 후 반환합니다.
                return u + solution(v);
            } else {
                // 4. 문자열 u가 "올바른 괄호 문자열"이 아니라면
                StringBuilder sb = new StringBuilder();
                
                // 4-1. 빈 문자열에 첫 번째 문자로 '('를 붙입니다.
                sb.append("(");
                // 4-2. 문자열 v에 대해 1단계부터 재귀적으로 수행한 결과 문자열을 이어 붙입니다.
                sb.append(solution(v));
                // 4-3. ')'를 다시 붙입니다.
                sb.append(")");
                
                // 4-4. u의 첫 번째와 마지막 문자를 제거하고, 나머지 문자열의 괄호 방향을 뒤집어서 뒤에 붙입니다.
                sb.append(reverseParentheses(u));
                
                // 4-5. 생성된 문자열을 반환합 니다.
                return sb.toString();
            }
        }
        
        // 균형잡힌 괄호 문자열 u의 분리 지점(인덱스)을 찾는 함수
        private int getSplitIndex(String str) {
            int count = 0;
            for (int i = 0; i < str.length(); i++) {
                if (str.charAt(i) == '(') {
                    count++;
                } else {
                    count--;
                }
                // '('와 ')'의 개수가 같아지는 시점(count == 0)에서 u의 끝 다음 인덱스를 반환
                if (count == 0) {
                    return i + 1;
                }
            }
            return str.length();
        }
        
        // 올바른 괄호 문자열인지 검증하는 함수
        private boolean isCorrect(String str) {
            int count = 0;
            for (int i = 0; i < str.length(); i++) {
                if (str.charAt(i) == '(') {
                    count++;
                } else {
                    count--;
                }
                // 닫는 괄호가 먼저 많이 나와 음수가 되는 순간 올바른 괄호가 아님
                if (count < 0) {
                    return false;
                }
            }
            return count == 0;
        }
        
        //  첫 번째/마지막 문자를 제거하고 괄호 방향을 뒤집는 함수 (4-4번 과정)
        private String reverseParentheses(String u) {
            StringBuilder sb = new StringBuilder();

            // 첫 문자와 마지막 문자 제외한 부문자열 순회
            for (int i = 1; i < u.length() - 1; i++) {
                if (u.charAt(i) == '(') {
                    sb.append(")");
                } else {
                    sb.append("(");
                }
            }
            return sb.toString();
        }
    }

# 문제 정의

### 제약 조건 분석

곡 수 최대 100개, 악보 및 멜로디 길이 최대 1,439개로 매우 작음. 재생 시간 계산과 문자열 치환/검색 알고리즘으로 시간 초과 없이 충분히 해결 가능.

---

### 패턴 단순화 및 추상화

- C#, D#, F#, G#, A# 등 샵(#)이 붙은 음: 소문자(c, d, f, g, a)나 단일 문자로 치환해 비교 단순화
- 시작/종료 시각(HH:MM): 분 단위 정수로 변환하여 실제 재생 시간 산출
- 실제 방송된 전체 멜로디: 재생 시간에 맞춰 생성 후 기억한 멜로디 포함 여부 확인

---

### 경계 조건

- 멜로디 뒤 '#' 처리: 'ABC' 검색 시 'ABC#'은 다른 음으로 판별 (음 치환으로 해결)
- 일치하는 음악 복수 존재: 재생 시간이 긴 음악 우선, 동일 시 먼저 입력된 음악 선택
- 일치하는 음악 부재: "(None)" 반환

---

### 최적의 알고리즘

1. 기억한 멜로디 m과 악보 정보의 샵(#) 음을 단일 문자로 치환

2. 각 음악의 시작/종료 시각을 분 단위로 변환해 총 재생 시간 계산

3. 재생 시간에 맞춰 악보를 반복 또는 잘라 실제 방송된 전체 멜로디 생성

4. 치환된 멜로디 m이 전체 멜로디에 포함되는지 확인

5. 조건 만족 곡 중 재생 시간이 가장 긴 곡 선택, 동일 시 먼저 등장한 곡 제목 반환


# 코드


    import java.util.ArrayList;
    import java.util.List;

    class Solution {
        public String solution(String m, String[] musicinfos) {
            // 샵 처리
            m = replaceMelody(m);
            
            String answerTitle = "(None)";
            int maxPlayTime = -1;
            
            for (String info : musicinfos) {
                String[] parts = info.split(",");
                String startTimeStr = parts[0];
                String endTimeStr = parts[1];
                String title = parts[2];
                String sheet = replaceMelody(parts[3]);
                
                // 재생 시간 계산 
                int playTime = getMinutes(endTimeStr) - getMinutes(startTimeStr);
                
                // 실제 재생된 전체 멜로디 생성
                StringBuilder fullMelody = new StringBuilder();
                int sheetLen = sheet.length();
                for (int i = 0; i < playTime; i++) {
                    fullMelody.append(sheet.charAt(i % sheetLen));
                }
                
                // 기억한 멜로디가 포함되어 있는지 확인
                if (fullMelody.toString().contains(m)) {
                    // 재생 시간이 긴 음악 우선 선택 (동일하면 먼저 등장한 곡 유지)
                    if (playTime > maxPlayTime) {
                        maxPlayTime = playTime;
                        answerTitle = title;
                    }
                }
            }
            
            return answerTitle;
        }

        // 분 단위 정수로 변환
        private int getMinutes(String time) {
            String[] t = time.split(":");
            return Integer.parseInt(t[0]) * 60 + Integer.parseInt(t[1]);
        }

        // 샵(#)이 포함된 음을 단일 문자로 치환
        private String replaceMelody(String melody) {
            return melody.replace("C#", "c")
                        .replace("D#", "d")
                        .replace("F#", "f")
                        .replace("G#", "g")
                        .replace("A#", "a")
                        .replace("B#", "b");
        }
    }
